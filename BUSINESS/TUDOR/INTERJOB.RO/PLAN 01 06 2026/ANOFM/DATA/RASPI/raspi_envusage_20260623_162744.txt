from datetime import datetime

env_file = Path('/opt/ACTIVE/ANOFM/.env')
if env_file.exists():
    with open(env_file) as f:
        for line in f:
            if line.startswith('export '):
                key, val = line.replace('export ', '').strip().split('=', 1)
                os.environ[key] = val.strip("'")

limit = int(sys.argv[1]) if len(sys.argv) > 1 else 150
delay = int(sys.argv[2]) if len(sys.argv) > 2 else 8

    'python3', 'campaign_anofm_angajatori.py',
    '--limit', str(limit),
    '--delay', str(delay),
    '--daily-cap', str(limit)
]

print(f'[{datetime.now().isoformat()}] Running: {" ".join(cmd)}')
result = subprocess.run(cmd, env=os.environ)
sys.exit(result.returncode)
